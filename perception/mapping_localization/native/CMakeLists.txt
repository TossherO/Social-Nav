cmake_minimum_required(VERSION 3.16)
project(social_nav_localization_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters kdtree registration io)
find_package(yaml-cpp REQUIRED)
set(FASTLIO ${CMAKE_CURRENT_SOURCE_DIR}/../backends/roll_fast_lio/fastlio-loc)
add_library(social_nav_lio SHARED frontend.cpp ${FASTLIO}/src/MotionInitializer.cpp
    ${FASTLIO}/include/ikd-Tree/ikd_Tree.cpp)
target_compile_definitions(social_nav_lio PRIVATE SOCIAL_NAV_NATIVE ROOT_DIR="" PCL_NO_PRECOMPILE)
target_compile_options(social_nav_lio PRIVATE -O2)
target_include_directories(social_nav_lio PRIVATE ${FASTLIO}/include ${FASTLIO}/src ${PCL_INCLUDE_DIRS})
target_link_libraries(social_nav_lio PRIVATE Eigen3::Eigen OpenMP::OpenMP_CXX ${PCL_LIBRARIES} yaml-cpp)

option(SOCIAL_NAV_BUILD_LIO_CUDA "Build optional FAST-LIO CUDA matching backend" OFF)
if(SOCIAL_NAV_BUILD_LIO_CUDA)
    if(NOT SOCIAL_NAV_CUDA_ARCH MATCHES "^[0-9]+$")
        message(FATAL_ERROR "Set SOCIAL_NAV_CUDA_ARCH to the target compute capability")
    endif()
    enable_language(CUDA)
    add_library(social_nav_lio_cuda SHARED cuda_matcher.cu)
    # CMake 3.16 does not recognize CUDA 13's C++17 dialect automatically.
    target_compile_options(social_nav_lio_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-std=c++17;--fmad=false;-arch=sm_${SOCIAL_NAV_CUDA_ARCH}>)
    target_compile_definitions(social_nav_lio PRIVATE SOCIAL_NAV_LIO_CUDA)
    target_link_libraries(social_nav_lio PRIVATE social_nav_lio_cuda)
    add_executable(social_nav_lio_cuda_test EXCLUDE_FROM_ALL ${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/native/lio_cuda.cpp)
    target_include_directories(social_nav_lio_cuda_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(social_nav_lio_cuda_test PRIVATE social_nav_lio_cuda)
endif()

find_package(GTSAM REQUIRED)
set(ROLL ${CMAKE_CURRENT_SOURCE_DIR}/../backends/roll_fast_lio/roll)
add_library(social_nav_roll SHARED roll.cpp)
target_compile_features(social_nav_roll PRIVATE cxx_std_17)
target_compile_definitions(social_nav_roll PRIVATE SOCIAL_NAV_NATIVE_ROLL)
target_compile_options(social_nav_roll PRIVATE -O2)
target_include_directories(social_nav_roll PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${ROLL}/include ${PCL_INCLUDE_DIRS} ${GTSAM_INCLUDE_DIR})
target_link_libraries(social_nav_roll PRIVATE Eigen3::Eigen OpenMP::OpenMP_CXX ${PCL_LIBRARIES} yaml-cpp gtsam)
if(SOCIAL_NAV_BUILD_LIO_CUDA)
    target_compile_definitions(social_nav_roll PRIVATE SOCIAL_NAV_LIO_CUDA)
    target_link_libraries(social_nav_roll PRIVATE social_nav_lio_cuda)
endif()
