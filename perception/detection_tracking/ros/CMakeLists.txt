cmake_minimum_required(VERSION 3.0.2)
project(social_nav_people)
find_package(catkin REQUIRED COMPONENTS rospy std_msgs sensor_msgs nav_msgs geometry_msgs visualization_msgs message_generation social_nav_localization)
add_message_files(FILES Detection.msg TrackState.msg People.msg ImageAssociation.msg ImageAssociations.msg)
generate_messages(DEPENDENCIES std_msgs)
catkin_package(CATKIN_DEPENDS rospy std_msgs sensor_msgs nav_msgs geometry_msgs visualization_msgs message_runtime social_nav_localization)
get_filename_component(SOCIAL_NAV_PEOPLE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
catkin_install_python(PROGRAMS scripts/people_node "${SOCIAL_NAV_PEOPLE_SOURCE}/../../scripts/associate_people_rgb.py" DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/launch)
