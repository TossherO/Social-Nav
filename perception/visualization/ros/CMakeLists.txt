cmake_minimum_required(VERSION 3.10)
project(social_nav_rviz)
find_package(catkin REQUIRED COMPONENTS roscpp rviz pluginlib geometry_msgs sensor_msgs)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
catkin_package(LIBRARIES social_nav_rviz_tools CATKIN_DEPENDS roscpp rviz pluginlib geometry_msgs sensor_msgs)
add_library(social_nav_rviz_tools src/ground_pose_tool.cpp)
target_compile_features(social_nav_rviz_tools PRIVATE cxx_std_14)
target_include_directories(social_nav_rviz_tools PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(social_nav_rviz_tools ${catkin_LIBRARIES} Qt5::Widgets)
install(TARGETS social_nav_rviz_tools LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  get_filename_component(SOCIAL_NAV_RVIZ_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
  add_executable(ground_pick_check "${SOCIAL_NAV_RVIZ_SOURCE}/../../../tests/test_rviz_ground_pick.cpp")
  target_include_directories(ground_pick_check PRIVATE "${SOCIAL_NAV_RVIZ_SOURCE}/src")
  target_compile_features(ground_pick_check PRIVATE cxx_std_14)
  add_test(NAME ground_pick_check COMMAND ground_pick_check)
endif()
