cmake_minimum_required(VERSION 3.16)
project(social_nav_environment_cuda LANGUAGES CXX CUDA)
if(NOT DEFINED SOCIAL_NAV_CUDA_ARCH)
  message(FATAL_ERROR "Set SOCIAL_NAV_CUDA_ARCH to the target compute capability, e.g. 86")
endif()
add_library(social_nav_occupancy_cuda SHARED occupancy_cuda.cu)
target_compile_features(social_nav_occupancy_cuda PRIVATE cxx_std_14)
target_compile_options(social_nav_occupancy_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-arch=sm_${SOCIAL_NAV_CUDA_ARCH};--fmad=false>)
set_target_properties(social_nav_occupancy_cuda PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
