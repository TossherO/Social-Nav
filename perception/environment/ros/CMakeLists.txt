cmake_minimum_required(VERSION 3.0.2)
project(social_nav_environment)
find_package(catkin REQUIRED COMPONENTS social_nav_localization nav_msgs sensor_msgs std_msgs message_generation)
find_package(octomap REQUIRED)
add_message_files(FILES EnvironmentGrid.msg)
add_service_files(FILES QueryEnvironment.srv)
generate_messages(DEPENDENCIES nav_msgs std_msgs)
catkin_package(LIBRARIES social_nav_occupancy CATKIN_DEPENDS social_nav_localization nav_msgs sensor_msgs std_msgs message_runtime)
get_filename_component(SOCIAL_NAV_ENV_ROS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
add_library(social_nav_occupancy SHARED ${SOCIAL_NAV_ENV_ROS_SOURCE}/../native/occupancy.cpp)
target_compile_features(social_nav_occupancy PRIVATE cxx_std_14)
target_include_directories(social_nav_occupancy PRIVATE ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(social_nav_occupancy ${OCTOMAP_LIBRARIES})
install(TARGETS social_nav_occupancy LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
catkin_install_python(PROGRAMS scripts/environment_node scripts/perception_scene DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
