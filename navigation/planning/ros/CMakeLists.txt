cmake_minimum_required(VERSION 3.10)
project(social_nav_planning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(ROS_NAVIGATION_PREFIX "" CACHE PATH "Optional extracted Noetic package prefix")
find_path(NAV_INCLUDE global_planner/dijkstra.h HINTS "${ROS_NAVIGATION_PREFIX}/include" /opt/ros/noetic/include REQUIRED)
find_library(NAV_LIBRARY global_planner HINTS "${ROS_NAVIGATION_PREFIX}/lib" /opt/ros/noetic/lib REQUIRED)
find_library(COSTMAP_LIBRARY costmap_2d HINTS "${ROS_NAVIGATION_PREFIX}/lib" /opt/ros/noetic/lib REQUIRED)
get_filename_component(SOCIAL_NAV_PLANNING_ROS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
add_library(social_nav_global_planner SHARED global_planner.cpp ${SOCIAL_NAV_PLANNING_ROS_SOURCE}/../native/pose_search.cpp)
target_include_directories(social_nav_global_planner PRIVATE ${NAV_INCLUDE} /opt/ros/noetic/include /usr/include/eigen3)
target_link_libraries(social_nav_global_planner PRIVATE ${NAV_LIBRARY} ${COSTMAP_LIBRARY})
get_filename_component(NAV_LIB_DIR "${NAV_LIBRARY}" DIRECTORY)
set_target_properties(social_nav_global_planner PROPERTIES BUILD_RPATH "${NAV_LIB_DIR};/opt/ros/noetic/lib")
target_link_options(social_nav_global_planner PRIVATE -Wl,--disable-new-dtags)
