cmake_minimum_required(VERSION 3.16)
project(social_nav_pose_search LANGUAGES CXX)
add_library(social_nav_pose_search SHARED pose_search.cpp)
target_compile_features(social_nav_pose_search PRIVATE cxx_std_14)
target_compile_options(social_nav_pose_search PRIVATE -O3)
option(SOCIAL_NAV_BUILD_CUDA "Build resident collision certification" OFF)
if(SOCIAL_NAV_BUILD_CUDA)
  if(NOT DEFINED SOCIAL_NAV_CUDA_ARCH)
    message(FATAL_ERROR "Set SOCIAL_NAV_CUDA_ARCH")
  endif()
  enable_language(CUDA)
  add_library(social_nav_collision_cuda SHARED collision.cu)
  target_compile_features(social_nav_collision_cuda PRIVATE cxx_std_14)
  target_compile_options(social_nav_collision_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-arch=sm_${SOCIAL_NAV_CUDA_ARCH};--fmad=false>)
  set_target_properties(social_nav_collision_cuda PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
endif()
